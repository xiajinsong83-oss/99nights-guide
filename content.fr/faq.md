[[faq]]
  q = "99 Nights in the Forest est-il gratuit ?"
  a = "Oui. C'est un jeu Roblox gratuit. Certaines classes et cosmétiques demandent des gemmes ou du Robux, mais le jeu principal et toute la progression sont gratuits."

[[faq]]
  q = "Sur quelles plateformes peut-on y jouer ?"
  a = "Partout où Roblox fonctionne : PC, Mac, iOS, Android, Xbox et VR. La progression est liée à ton compte Roblox."

[[faq]]
  q = "À combien de joueurs peut-on jouer ?"
  a = "C'est un jeu coopératif conçu pour les groupes. La capacité des serveurs varie, mais le jeu est pensé pour que des groupes de 2–4 ou plus coopèrent."

[[faq]]
  q = "Peut-on jouer à 99 Nights in the Forest en solo ?"
  a = "Oui, mais c'est nettement plus dur. Les solos devraient rallonger la phase « essentiel » du début, ne pas rôder la nuit et compter sur les pièges et le kite plutôt que le combat frontal."

[[faq]]
  q = "Comment jouer avec des amis ?"
  a = "Entre dans le même serveur Roblox : rejoins la partie d'un ami depuis la liste d'amis Roblox ou utilise le code de serveur/lien proposé par le jeu. Ensuite, construisez le camp ensemble."

[[faq]]
  q = "99 Nights in the Forest est-il basé sur le film ?"

  q = "Comment sauver les enfants disparus ?"
