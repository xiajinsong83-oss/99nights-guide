[[faq]]
  q = "99 Nights in the Forest は無料で遊べますか？"
  a = "はい。無料の Roblox ゲームです。一部のクラスとスキンはジェムか Robux が必要ですが、コアゲームとすべての進行は無料です。"

[[faq]]
  q = "どんなプラットフォームで遊べますか？"
  a = "Roblox が動く場所ならどこでも：PC、Mac、iOS、Android、Xbox、VR。進行は Roblox アカウントに紐づきます。"

[[faq]]
  q = "何人で一緒に遊べますか？"
  a = "チーム向けに設計された協力ゲームです。サーバー容量は異なりますが、ゲームは2〜4人以上のグループが協力する前提で作られています。"

[[faq]]
  q = "99 Nights in the Forest はソロで遊べますか？"
  a = "できますが、かなり難しくなります。ソロは序盤の「コア」フェーズを延ばし、夜にうろつかず、正面戦闘より罠とカイトを頼りにしましょう。"

[[faq]]
  q = "友達とどう遊びますか？"
  a = "同じ Roblox サーバーに入ります：Roblox のフレンドリストから友達のゲームに参加するか、ゲームが提供するサーバーコード／参加リンクを使います。それから一緒にキャンプを建てましょう。"

[[faq]]
  q = "99 Nights in the Forest は映画がベースですか？"
  a = "20世紀スタジオの映画とのタイアップがあり、多くの新規プレイヤーを呼び込んでいます。ゲーム自体は Roblox の協力型サバイバルホラーです。"

[[faq]]
  q = "行方不明の子供をどう救出しますか？"
  a = "食料、武器、光源を持って系統的に探索します。マップに手がかりをマーク；手がかりを見つけて追うと、救出ランと後半コンテンツが解放されます。"

[[faq]]
  q = "コードは大文字小文字を区別しますか？"
